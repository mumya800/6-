//1.Инициализация массива строк lines:

   char lines[3][30] = {"/test.txt", "Привет", "Мир"};
   
//Здесь создаётся двумерный массив строк, где:
   - lines[0] — имя файла ("/test.txt"),
   - lines[1] — первая строка содержимого ("Привет"),
   - lines[2] — вторая строка содержимого ("Мир").

//2.Вывод имени файла:

   printf("Имя файла: %s\n", lines[0]);
   
//Просто выводит первую строку массива — имя файла.

//3. Формирование строки content из строк с индексами 1 и 2:

   char content[100];
   int index = 0;
   for (int i = 1; i < 3; i++) 
   {
       int j = 0;
       while (lines[i][j] != '\0') {
           content[index] = lines[i][j];
           index++;
           j++;
       }
       content[index] = '\n'; 
       index++;
   }
   content[index] = '\0';
   
/*Цикл проходит по строкам с индексами 1 и 2 ("Привет" и "Мир").
Внутренний цикл копирует символы из каждой строки в массив content.
После каждой строки добавляется символ новой строки '\n'.
В конце добавляется нулевой символ '\0', чтобы обозначить конец строки.*/
   
//В итоге переменная content содержит:
   
   Привет\nМир\n\0

//4. Вывод содержимого "файла":

   printf("\nСодержимое файла:\n%s", content);
   
   
//5. Поиск подстроки "Привет" в содержимом:

//Сначала вычисляются длины строк content и target:
   
   char target[] = "Привет";
   
   int contentLength = 0;
   while (content[contentLength] != '\0') {
       contentLength++;
   }

   int targetLength = 0;
   while (target[targetLength] != '\0') {
       targetLength++;
   }

//6. Поиск подстроки методом перебора:
   
   int found = 0;
   
   for (int i = 0; i <= contentLength - targetLength; i++) {
       int match = 1;
       for (int j = 0; j < targetLength; j++) {
           if (content[i + j] != target[j]) {
               match = 0;
               break;
           }
       }
       if (match == 1) {
           found = 1;
           break;
       }
   }
   
/*Внешний цикл перебирает все возможные позиции в строке content, где может начинаться подстрока.
Внутренний цикл сравнивает символы подстроки с соответствующими символами в content.
Если все символы совпали, устанавливается флаг found=1, и поиск прерывается.*/

//7. Вывод результата поиска:
   
if (found) 
{
    printf("\nСлово \"%s\" найдено в тексте\n", target);
} 
else 
{
    printf("\nСлово \"%s\" не найдено\n", target);
}

- Если слово найдено, сообщает об этом, иначе — что слово не найдено.
